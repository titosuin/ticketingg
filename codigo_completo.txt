import type {
  Ticket,
  TicketState,
  Severity,
  EnqueueTicketInput,
  RequeueStrategy,
  QueueMetrics,
  AuditTransition,
} from './types.ts';
import { generateUUIDv7 } from './uuidv7.ts';
import { AsyncMutex } from './concurrency.ts';
import { IndexedPriorityQueue } from './priorityQueue.ts';

export interface EngineConfig {
  defaultLeaseTtlMs?: number;
  defaultMaxRetries?: number;
  agingRatePerMs?: number;
  slaUrgencyWeight?: number;
  baseBackoffMs?: number;
}

export class TicketEngine {
  private queue: IndexedPriorityQueue;
  private processingTickets: Map<string, Ticket> = new Map();
  private completedTickets: Map<string, Ticket> = new Map();
  private deadLetterTickets: Map<string, Ticket> = new Map();
  private allTickets: Map<string, Ticket> = new Map();

  private mutex: AsyncMutex = new AsyncMutex();
  private config: Required<EngineConfig>;

  // Running telemetry accumulators for O(1) metrics
  private totalCreatedCounter = 0;
  private totalCompletedDwellSumMs = 0;
  private totalCompletedCount = 0;
  private requeuedEventCount = 0;

  constructor(config: EngineConfig = {}) {
    this.config = {
      defaultLeaseTtlMs: config.defaultLeaseTtlMs ?? 5000, // 5 seconds default TTL
      defaultMaxRetries: config.defaultMaxRetries ?? 3,
      agingRatePerMs: config.agingRatePerMs ?? 0.05,
      slaUrgencyWeight: config.slaUrgencyWeight ?? 0.02,
      baseBackoffMs: config.baseBackoffMs ?? 2000,
    };

    this.queue = new IndexedPriorityQueue({
      agingRatePerMs: this.config.agingRatePerMs,
      slaWeight: this.config.slaUrgencyWeight,
    });
  }

  /**
   * Enqueue a new ticket into the engine.
   * Time Complexity: O(log N)
   */
  public async enqueue(
    input: EnqueueTicketInput,
    customPriority?: number,
    now: number = Date.now()
  ): Promise<Ticket> {
    return this.mutex.withLock(() => {
      const severity: Severity = input.severity ?? 2;
      const baseScores: Record<Severity, number> = {
        1: 100, // Low
        2: 250, // Medium
        3: 500, // High
        4: 1000, // Critical
      };

      const basePriorityScore = customPriority ?? input.customBaseScore ?? baseScores[severity];
      const maxRetries = input.maxRetries ?? this.config.defaultMaxRetries;
      const slaMinutes = input.slaMinutes ?? (severity === 4 ? 15 : severity === 3 ? 60 : severity === 2 ? 240 : 1440);
      const slaDeadlineMs = now + slaMinutes * 60_000;

      const ticketId = generateUUIDv7(now);

      const initialAudit: AuditTransition = {
        transitionId: generateUUIDv7(now),
        fromState: null,
        toState: 'QUEUED',
        timestamp: now,
        actor: input.author.id,
        reason: 'Ticket created and enqueued into priority queue',
        metadata: {
          channel: input.author.channel,
          severity,
          basePriorityScore,
          slaDeadlineMs,
        },
      };

      const ticket: Ticket = {
        id: ticketId,
        state: 'QUEUED',
        author: input.author,
        payload: {
          title: input.title,
          description: input.description,
          tags: input.tags,
          context: input.context,
        },
        severity,
        basePriorityScore,
        slaDeadlineMs,
        timestamps: {
          createdAt: now,
          enqueuedAt: now,
          assignedAt: null,
          resolvedAt: null,
          lastEvictionAt: null,
          leaseExpiresAt: null,
          backoffUntil: null,
        },
        dispatchTelemetry: {
          retryCount: 0,
          maxRetries,
          currentWorkerId: null,
          leaseTtlMs: this.config.defaultLeaseTtlMs,
          auditTrail: [initialAudit],
        },
      };

      this.allTickets.set(ticket.id, ticket);
      this.queue.enqueue(ticket, now);
      this.totalCreatedCounter++;

      return { ...ticket };
    });
  }

  /**
   * Peek at a window of tickets in the queue ordered by current effective priority: O(K log K)
   */
  public async peek_queue(
    limit: number = 10,
    offset: number = 0,
    now: number = Date.now()
  ): Promise<Ticket[]> {
    return this.mutex.withLock(() => {
      // First evict any expired leases
      this.internalEvictExpiredLeases(now);
      return this.queue.peekRange(limit, offset, now);
    });
  }

  /**
   * Atomic Lease Acquisition Protocol.
   * Extracts the highest priority eligible ticket and issues a time-bound lease to worker_id.
   * Guarantees race-condition freedom across concurrent worker threads.
   * Time Complexity: O(log N)
   */
  public async acquire_next(
    worker_id: string,
    lease_ttl_ms?: number,
    now: number = Date.now()
  ): Promise<Ticket | null> {
    return this.mutex.withLock(() => {
      // 1. Automatic housekeeping: evict expired worker leases back into queue
      this.internalEvictExpiredLeases(now);

      const effectiveTtl = lease_ttl_ms ?? this.config.defaultLeaseTtlMs;

      // 2. Extract next eligible candidate (handling backoff schedules)
      const ticket = this.extractEligibleTicket(now);
      if (!ticket) {
        return null;
      }

      // 3. FSM Validation: QUEUED -> PROCESSING
      if (ticket.state !== 'QUEUED') {
        throw new Error(
          `FSM Invariant Violation: Attempted to acquire ticket ${ticket.id} from invalid state '${ticket.state}'. Expected 'QUEUED'.`
        );
      }

      // 4. Atomic Lease Assignment
      ticket.state = 'PROCESSING';
      ticket.dispatchTelemetry.currentWorkerId = worker_id;
      ticket.dispatchTelemetry.leaseTtlMs = effectiveTtl;
      ticket.timestamps.assignedAt = now;
      ticket.timestamps.leaseExpiresAt = now + effectiveTtl;

      // 5. Audit Trail Transition Record
      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'QUEUED',
        toState: 'PROCESSING',
        timestamp: now,
        actor: worker_id,
        reason: `Atomic lease acquired with TTL of ${effectiveTtl}ms`,
        metadata: {
          workerId: worker_id,
          leaseExpiresAt: ticket.timestamps.leaseExpiresAt,
          leaseTtlMs: effectiveTtl,
        },
      });

      this.processingTickets.set(ticket.id, ticket);
      return { ...ticket };
    });
  }

  /**
   * Commits the definitive resolution of a ticket.
   * Transitions ticket from PROCESSING to COMPLETED.
   * Time Complexity: O(1)
   */
  public async commit_resolution(
    ticket_id: string,
    worker_id: string,
    resolution_notes: string,
    closing_metadata?: Record<string, unknown>,
    now: number = Date.now()
  ): Promise<Ticket> {
    return this.mutex.withLock(() => {
      const ticket = this.processingTickets.get(ticket_id);
      if (!ticket) {
        // Check if ticket is in completed or DLQ or was evicted
        const existing = this.allTickets.get(ticket_id);
        if (!existing) {
          throw new Error(`Ticket ${ticket_id} not found.`);
        }
        throw new Error(
          `Cannot commit resolution: Ticket ${ticket_id} is in state '${existing.state}', not 'PROCESSING'.`
        );
      }

      // Verify Worker Identity (Prevent unauthorized commit)
      if (ticket.dispatchTelemetry.currentWorkerId !== worker_id) {
        throw new Error(
          `Lease Conflict: Ticket ${ticket_id} is leased to '${ticket.dispatchTelemetry.currentWorkerId}', but worker '${worker_id}' attempted commit.`
        );
      }

      // Verify Lease Expiration
      if (ticket.timestamps.leaseExpiresAt && now > ticket.timestamps.leaseExpiresAt) {
        // Lease has expired! Reclaim and evict immediately
        this.internalEvictSingleLease(ticket, now, `Worker commit attempted after lease expiration`);
        throw new Error(
          `Lease Expired: Worker '${worker_id}' lease on ticket ${ticket_id} expired at ${ticket.timestamps.leaseExpiresAt} (now: ${now}). Ticket was evicted.`
        );
      }

      // FSM Transition: PROCESSING -> COMPLETED
      ticket.state = 'COMPLETED';
      ticket.timestamps.resolvedAt = now;
      ticket.resolutionNotes = resolution_notes;
      ticket.closingMetadata = closing_metadata;

      // Calculate dwell time metrics
      const totalLifecycleDwell = now - ticket.timestamps.createdAt;
      this.totalCompletedDwellSumMs += totalLifecycleDwell;
      this.totalCompletedCount++;

      // Audit entry
      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'PROCESSING',
        toState: 'COMPLETED',
        timestamp: now,
        actor: worker_id,
        reason: 'Ticket successfully resolved and committed',
        metadata: {
          resolutionNotes: resolution_notes,
          closingMetadata: closing_metadata,
          lifecycleDwellMs: totalLifecycleDwell,
        },
      });

      this.processingTickets.delete(ticket_id);
      this.completedTickets.set(ticket_id, ticket);

      return { ...ticket };
    });
  }

  /**
   * Requeues an in-flight ticket (e.g. worker transient failure, rejection, or backoff).
   * Transitions PROCESSING -> REQUEUED -> QUEUED (or DEAD_LETTER if retries exhausted).
   * Strategies:
   * - 'HEAD': Priority boost (LIFO burst) for immediate remedial processing.
   * - 'TAIL': Enqueued with normal priority at tail.
   * - 'EXPONENTIAL_BACKOFF': Deferred by 2^(retries-1) * baseBackoffMs.
   * Time Complexity: O(log N)
   */
  public async requeue(
    ticket_id: string,
    reason: string,
    strategy: RequeueStrategy = 'EXPONENTIAL_BACKOFF',
    now: number = Date.now()
  ): Promise<Ticket> {
    return this.mutex.withLock(() => {
      const ticket = this.processingTickets.get(ticket_id);
      if (!ticket) {
        const existing = this.allTickets.get(ticket_id);
        if (!existing) {
          throw new Error(`Ticket ${ticket_id} not found.`);
        }
        throw new Error(
          `Cannot requeue: Ticket ${ticket_id} is in state '${existing.state}', not 'PROCESSING'.`
        );
      }

      this.processingTickets.delete(ticket_id);
      this.requeuedEventCount++;

      // Increment retry counter
      ticket.dispatchTelemetry.retryCount++;
      const currentRetries = ticket.dispatchTelemetry.retryCount;
      const maxRetries = ticket.dispatchTelemetry.maxRetries;

      // Step 1: Transition to REQUEUED state
      ticket.state = 'REQUEUED';
      ticket.failureReason = reason;

      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'PROCESSING',
        toState: 'REQUEUED',
        timestamp: now,
        actor: ticket.dispatchTelemetry.currentWorkerId ?? 'SYSTEM',
        reason,
        metadata: {
          retryCount: currentRetries,
          maxRetries,
          strategy,
        },
      });

      // Clear current worker lock
      ticket.dispatchTelemetry.currentWorkerId = null;
      ticket.timestamps.leaseExpiresAt = null;

      // Step 2: Evaluate Dead Letter Queue condition
      if (currentRetries >= maxRetries) {
        // Exceeded max retries -> Isolate to DEAD_LETTER
        ticket.state = 'DEAD_LETTER';
        ticket.dispatchTelemetry.auditTrail.push({
          transitionId: generateUUIDv7(now),
          fromState: 'REQUEUED',
          toState: 'DEAD_LETTER',
          timestamp: now,
          actor: 'SYSTEM_DLQ_ROUTER',
          reason: `Exceeded maximum retry limit (${currentRetries}/${maxRetries}). Isolated for forensic audit.`,
          metadata: {
            fatalReason: reason,
            totalRetries: currentRetries,
          },
        });

        this.deadLetterTickets.set(ticket.id, ticket);
        return { ...ticket };
      }

      // Step 3: Re-enter QUEUED with chosen strategy
      ticket.state = 'QUEUED';
      ticket.timestamps.enqueuedAt = now;

      if (strategy === 'HEAD') {
        // High priority boost to jump to head of queue
        ticket.basePriorityScore += 2000;
        ticket.timestamps.backoffUntil = null;
      } else if (strategy === 'TAIL') {
        // Standard placement without boost
        ticket.timestamps.backoffUntil = null;
      } else if (strategy === 'EXPONENTIAL_BACKOFF') {
        // Delay release: baseBackoff * 2^(retryCount - 1)
        const delayMs = this.config.baseBackoffMs * Math.pow(2, currentRetries - 1);
        ticket.timestamps.backoffUntil = now + delayMs;
      }

      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'REQUEUED',
        toState: 'QUEUED',
        timestamp: now,
        actor: 'SYSTEM_DISPATCH',
        reason: `Re-enqueued with strategy: ${strategy}`,
        metadata: {
          strategy,
          backoffUntil: ticket.timestamps.backoffUntil,
          newBasePriority: ticket.basePriorityScore,
        },
      });

      this.queue.enqueue(ticket, now);
      return { ...ticket };
    });
  }

  /**
   * Heartbeat to extend active worker lease TTL: O(1)
   */
  public async heartbeat_lease(
    ticket_id: string,
    worker_id: string,
    extension_ms?: number,
    now: number = Date.now()
  ): Promise<Ticket> {
    return this.mutex.withLock(() => {
      const ticket = this.processingTickets.get(ticket_id);
      if (!ticket) {
        throw new Error(`Ticket ${ticket_id} is not currently in PROCESSING.`);
      }

      if (ticket.dispatchTelemetry.currentWorkerId !== worker_id) {
        throw new Error(`Worker '${worker_id}' does not own active lease for ${ticket_id}.`);
      }

      const ext = extension_ms ?? this.config.defaultLeaseTtlMs;
      ticket.timestamps.leaseExpiresAt = now + ext;

      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'PROCESSING',
        toState: 'PROCESSING',
        timestamp: now,
        actor: worker_id,
        reason: `Lease extended by ${ext}ms (Heartbeat)`,
        metadata: { newExpiresAt: ticket.timestamps.leaseExpiresAt },
      });

      return { ...ticket };
    });
  }

  /**
   * Periodic or on-demand eviction of expired worker leases.
   */
  public async evict_expired_leases(now: number = Date.now()): Promise<number> {
    return this.mutex.withLock(() => {
      return this.internalEvictExpiredLeases(now);
    });
  }

  /**
   * Replay a ticket from the Dead Letter Queue back into active circulation.
   */
  public async replay_dlq_ticket(
    ticket_id: string,
    reset_retries: boolean = true,
    now: number = Date.now()
  ): Promise<Ticket> {
    return this.mutex.withLock(() => {
      const ticket = this.deadLetterTickets.get(ticket_id);
      if (!ticket) {
        throw new Error(`Ticket ${ticket_id} not found in Dead Letter Queue.`);
      }

      this.deadLetterTickets.delete(ticket_id);

      if (reset_retries) {
        ticket.dispatchTelemetry.retryCount = 0;
      }

      ticket.state = 'QUEUED';
      ticket.timestamps.enqueuedAt = now;
      ticket.timestamps.backoffUntil = null;

      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'DEAD_LETTER',
        toState: 'QUEUED',
        timestamp: now,
        actor: 'ADMIN_REPLAY',
        reason: 'Forensic operator replayed ticket back into active dispatch queue',
        metadata: { resetRetries: reset_retries },
      });

      this.queue.enqueue(ticket, now);
      return { ...ticket };
    });
  }

  /**
   * Retrieves high-frequency O(1) telemetry and metrics snapshot.
   */
  public get_metrics(now: number = Date.now()): QueueMetrics {
    const totalQueued = this.queue.size();
    const severityCounts = this.queue.getSeverityCounts();
    const { totalDwellMs, oldestDwellMs } = this.queue.getTotalCurrentDwellMs(now);

    const avgDwell = totalQueued > 0 ? Math.round(totalDwellMs / totalQueued) : 0;

    // Active leases list with countdown
    const activeLeases = Array.from(this.processingTickets.values()).map((t) => ({
      ticketId: t.id,
      workerId: t.dispatchTelemetry.currentWorkerId ?? 'UNKNOWN',
      remainingMs: Math.max(0, (t.timestamps.leaseExpiresAt ?? now) - now),
      expiresAt: t.timestamps.leaseExpiresAt ?? now,
      severity: t.severity,
    }));

    // Count tickets nearing or breached SLA
    let slaBreachRiskCount = 0;
    for (const t of this.queue.getAll()) {
      if (t.slaDeadlineMs - now < 300_000) {
        // Less than 5 minutes to breach or already breached
        slaBreachRiskCount++;
      }
    }

    return {
      totalQueued,
      processingCount: this.processingTickets.size,
      completedCount: this.completedTickets.size,
      requeuedCount: this.requeuedEventCount,
      deadLetterCount: this.deadLetterTickets.size,
      totalCreated: this.totalCreatedCounter,
      countBySeverity: severityCounts,
      avgDwellTimeMs: avgDwell,
      totalCompletedDwellSumMs: this.totalCompletedDwellSumMs,
      oldestQueuedDwellMs: oldestDwellMs,
      slaBreachRiskCount,
      activeLeases,
    };
  }

  public get_ticket(id: string): Ticket | null {
    const t = this.allTickets.get(id);
    return t ? { ...t } : null;
  }

  public get_all_tickets(): Ticket[] {
    return Array.from(this.allTickets.values()).map((t) => ({ ...t }));
  }

  public get_dlq_tickets(): Ticket[] {
    return Array.from(this.deadLetterTickets.values()).map((t) => ({ ...t }));
  }

  public get_completed_tickets(): Ticket[] {
    return Array.from(this.completedTickets.values()).map((t) => ({ ...t }));
  }

  public get_processing_tickets(): Ticket[] {
    return Array.from(this.processingTickets.values()).map((t) => ({ ...t }));
  }

  // --- Internal Helper Methods ---

  private extractEligibleTicket(now: number): Ticket | null {
    const deferredList: Ticket[] = [];
    let chosen: Ticket | null = null;

    while (!this.queue.isEmpty()) {
      const top = this.queue.extractMax(now);
      if (!top) break;

      // Check if ticket is still under exponential backoff delay
      if (top.timestamps.backoffUntil && top.timestamps.backoffUntil > now) {
        deferredList.push(top);
      } else {
        chosen = top;
        break;
      }
    }

    // Re-insert deferred tickets back into queue
    for (const deferred of deferredList) {
      this.queue.enqueue(deferred, now);
    }

    return chosen;
  }

  private internalEvictExpiredLeases(now: number): number {
    let evictedCount = 0;
    for (const ticket of this.processingTickets.values()) {
      if (ticket.timestamps.leaseExpiresAt && now > ticket.timestamps.leaseExpiresAt) {
        this.internalEvictSingleLease(
          ticket,
          now,
          `Lease TTL of ${ticket.dispatchTelemetry.leaseTtlMs}ms expired without worker heartbeat`
        );
        evictedCount++;
      }
    }
    return evictedCount;
  }

  private internalEvictSingleLease(ticket: Ticket, now: number, reason: string): void {
    const workerId = ticket.dispatchTelemetry.currentWorkerId;
    this.processingTickets.delete(ticket.id);

    ticket.timestamps.lastEvictionAt = now;
    ticket.dispatchTelemetry.retryCount++;
    const retries = ticket.dispatchTelemetry.retryCount;
    const maxRetries = ticket.dispatchTelemetry.maxRetries;

    ticket.dispatchTelemetry.auditTrail.push({
      transitionId: generateUUIDv7(now),
      fromState: 'PROCESSING',
      toState: 'REQUEUED',
      timestamp: now,
      actor: 'SYSTEM_LEASE_EVICTOR',
      reason,
      metadata: {
        abandonedByWorker: workerId,
        retryCount: retries,
        maxRetries,
      },
    });

    ticket.dispatchTelemetry.currentWorkerId = null;
    ticket.timestamps.leaseExpiresAt = null;

    if (retries >= maxRetries) {
      ticket.state = 'DEAD_LETTER';
      ticket.dispatchTelemetry.auditTrail.push({
        transitionId: generateUUIDv7(now),
        fromState: 'REQUEUED',
        toState: 'DEAD_LETTER',
        timestamp: now,
        actor: 'SYSTEM_DLQ_ROUTER',
        reason: `Exceeded max retries after lease eviction (${retries}/${maxRetries})`,
      });
      this.deadLetterTickets.set(ticket.id, ticket);
    } else {
      ticket.state = 'QUEUED';
      ticket.timestamps.enqueuedAt = now;
      // Head boost to prioritize recovery
      ticket.basePriorityScore += 500;
      this.queue.enqueue(ticket, now);
    }
  }
}

/**
 * Monotonically increasing UUIDv7 implementation according to RFC 9562.
 * UUIDv7 embeds a 48-bit millisecond timestamp at the front, guaranteeing
 * temporal natural ordering, index locality, and collision resistance.
 */

let lastTimestamp = -1;
let subMsCounter = 0;

export function generateUUIDv7(customTimestamp?: number): string {
  const now = customTimestamp ?? Date.now();

  if (now === lastTimestamp) {
    subMsCounter = (subMsCounter + 1) & 0x0fff; // 12-bit sequence counter
    if (subMsCounter === 0) {
      // Counter overflow within same millisecond: artificially step timestamp forward
      lastTimestamp++;
    }
  } else {
    lastTimestamp = now;
    // Initialize counter with random 4-bit seed to deter predictability
    subMsCounter = Math.floor(Math.random() * 0x10);
  }

  const effectiveTimestamp = Math.max(now, lastTimestamp);

  // 48-bit timestamp
  const timeHex = effectiveTimestamp.toString(16).padStart(12, '0');
  const timeHigh = timeHex.slice(0, 8);
  const timeMid = timeHex.slice(8, 12);

  // 4-bit version (7) + 12-bit sequence/random counter
  const verAndSeq = ((7 << 12) | (subMsCounter & 0x0fff)).toString(16).padStart(4, '0');

  // 2-bit variant (10xx) + 14-bit random
  const randA = (0x8000 | (Math.floor(Math.random() * 0x3fff))).toString(16).padStart(4, '0');

  // 48-bit trailing randomness
  const randB = Array.from({ length: 6 }, () =>
    Math.floor(Math.random() * 256)
      .toString(16)
      .padStart(2, '0')
  ).join('');

  return `${timeHigh}-${timeMid}-${verAndSeq}-${randA}-${randB}`.toLowerCase();
}

/**
 * Extracts Unix millisecond timestamp from canonical UUIDv7 string.
 */
export function extractTimestampFromUUIDv7(uuid: string): number {
  const clean = uuid.replace(/-/g, '');
  if (clean.length < 12) return NaN;
  const timeHex = clean.slice(0, 12);
  return parseInt(timeHex, 16);
}

test('UUIDv7: generates RFC 9562 compliant monotonically sortable identifiers', () => {
  const ids: string[] = [];
  const t0 = Date.now();
  for (let i = 0; i < 50; i++) {
    ids.push(generateUUIDv7(t0 + i));
  }

  // 1. Regex check for UUID format
  const uuidRegex = /^[0-9a-f]{8}-[0-9a-f]{4}-7[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;
  for (const id of ids) {
    assert.match(id, uuidRegex, `ID ${id} must match RFC 9562 UUIDv7 format`);
  }

  // 2. Monotonic sortability
  const sorted = [...ids].sort();
  assert.deepEqual(ids, sorted, 'UUIDv7 must be lexicographically monotonically sorted');

  // 3. Extracted timestamp sanity
  const extracted = extractTimestampFromUUIDv7(ids[0]);
  assert.equal(extracted, t0, 'Extracted timestamp must match generator input');
});

test('High-Concurrency Stress Test: 5 Concurrent Workers, 50 Tickets, Zero Race Conditions', async () => {
  const engine = new TicketEngine({
    defaultLeaseTtlMs: 200,
    defaultMaxRetries: 3,
    baseBackoffMs: 50,
  });

  const TOTAL_TICKETS = 50;
  const WORKER_COUNT = 5;

  // 1. Enqueue 50 mixed tickets
  const enqueuedTickets: Ticket[] = [];
  for (let i = 0; i < TOTAL_TICKETS; i++) {
    const sev = ((i % 4) + 1) as 1 | 2 | 3 | 4;
    const t = await engine.enqueue({
      title: `Stress Ticket #${i + 1}`,
      description: `Synthetic load test ticket index ${i}`,
      tags: ['stress', `batch-${Math.floor(i / 10)}`],
      severity: sev,
      author: {
        id: `usr_${i}`,
        name: `User ${i}`,
        department: i % 2 === 0 ? 'Engineering' : 'Support',
        channel: 'API',
      },
      maxRetries: 3,
    });
    enqueuedTickets.push(t);
  }

  assert.equal(engine.get_metrics().totalQueued, TOTAL_TICKETS);

  // 2. Concurrency tracking structures
  const acquiredCountPerTicket: Map<string, number> = new Map();
  const workerResolvedCount: Map<string, number> = new Map();
  const workerRequeuedCount: Map<string, number> = new Map();

  for (let w = 1; w <= WORKER_COUNT; w++) {
    workerResolvedCount.set(`worker-${w}`, 0);
    workerRequeuedCount.set(`worker-${w}`, 0);
  }

  // 3. Worker simulator
  const runWorker = async (workerId: string) => {
    let idleRounds = 0;
    while (idleRounds < 15) {
      const ticket = await engine.acquire_next(workerId, 300);
      if (!ticket) {
        idleRounds++;
        await new Promise((resolve) => setTimeout(resolve, 15));
        continue;
      }

      idleRounds = 0;

      // Assert no two workers ever acquire the same ticket concurrently
      const currentCount = (acquiredCountPerTicket.get(ticket.id) ?? 0) + 1;
      acquiredCountPerTicket.set(ticket.id, currentCount);

      // Simulate work duration
      await new Promise((resolve) => setTimeout(resolve, 5 + Math.floor(Math.random() * 15)));

      // Probabilistic decision: 70% resolve, 20% transient fail, 10% timeout crash
      const rand = Math.random();
      if (rand < 0.70) {
        await engine.commit_resolution(ticket.id, workerId, `Resolved by ${workerId}`);
        workerResolvedCount.set(workerId, (workerResolvedCount.get(workerId) ?? 0) + 1);
      } else if (rand < 0.90) {
        await engine.requeue(ticket.id, `Simulated transient error by ${workerId}`, 'EXPONENTIAL_BACKOFF');
        workerRequeuedCount.set(workerId, (workerRequeuedCount.get(workerId) ?? 0) + 1);
      } else {
        // Crash: abandon ticket and let lease expire; wait for eviction
        await new Promise((resolve) => setTimeout(resolve, 350));
        await engine.evict_expired_leases();
      }
    }
  };

  // Launch 5 concurrent workers
  const workerPromises = Array.from({ length: WORKER_COUNT }, (_, i) => runWorker(`worker-${i + 1}`));
  await Promise.all(workerPromises);

  // Final eviction sweep
  await engine.evict_expired_leases();

  const finalMetrics = engine.get_metrics();

  // Invariant 1: Ticket Conservation Law
  // Total Created === (Queued + Processing + Completed + DeadLetter)
  const accountedTickets =
    finalMetrics.totalQueued +
    finalMetrics.processingCount +
    finalMetrics.completedCount +
    finalMetrics.deadLetterCount;

  assert.equal(
    accountedTickets,
    TOTAL_TICKETS,
    `Ticket Conservation Invariant must hold: ${accountedTickets} accounted vs ${TOTAL_TICKETS} created`
  );

  // Invariant 2: Completed + DLQ + remaining active must equal total
  assert.equal(finalMetrics.totalCreated, TOTAL_TICKETS);
  assert.ok(finalMetrics.completedCount > 0, 'At least some tickets should be completed');

  // Invariant 3: Audit Trail Integrity
  for (const t of engine.get_all_tickets()) {
    const fresh = engine.get_ticket(t.id)!;
    assert.ok(fresh.dispatchTelemetry.auditTrail.length >= 1);
    // Chronological order verification
    for (let j = 1; j < fresh.dispatchTelemetry.auditTrail.length; j++) {
      assert.ok(
        fresh.dispatchTelemetry.auditTrail[j].timestamp >= fresh.dispatchTelemetry.auditTrail[j - 1].timestamp,
        `Audit trail for ${fresh.id} must be monotonically non-decreasing in time`
      );
    }
  }
});

export interface PriorityCalculatorOptions {
  agingRatePerMs?: number; // Rate at which queued tickets gain priority per ms (Starvation Prevention)
  slaWeight?: number; // Multiplier for SLA urgency countdown
}

/**
 * Calculates the dynamic effective score of a ticket at a given point in time.
 * Implements Starvation Prevention via monotonic aging + SLA deadline proximity boost.
 *
 * Formula:
 * Score(t) = BaseScore + (t - EnqueuedAt) * AgingRate + SLA_Urgency_Factor
 */
export function calculateDynamicScore(
  ticket: Ticket,
  now: number = Date.now(),
  options: PriorityCalculatorOptions = {}
): number {
  const agingRate = options.agingRatePerMs ?? 0.05; // 50 score points per second
  const slaWeight = options.slaWeight ?? 0.02;

  // 1. Monotonic aging component (prevents low-priority starvation)
  const dwellTimeMs = Math.max(0, now - ticket.timestamps.enqueuedAt);
  const agingBoost = dwellTimeMs * agingRate;

  // 2. SLA countdown urgency factor
  const timeToSlaMs = ticket.slaDeadlineMs - now;
  let slaBoost = 0;
  if (timeToSlaMs <= 0) {
    // SLA Breached: immediate maximum escalation boost
    slaBoost = 5000 + Math.abs(timeToSlaMs) * slaWeight;
  } else if (timeToSlaMs < 60_000) {
    // Under 1 minute to breach
    slaBoost = (60_000 - timeToSlaMs) * slaWeight * 2;
  }

  // 3. Penalty if under backoff delay
  const isBackoff = ticket.timestamps.backoffUntil && ticket.timestamps.backoffUntil > now;
  const backoffPenalty = isBackoff ? 1_000_000 : 0;

  const totalScore = ticket.basePriorityScore + agingBoost + slaBoost - backoffPenalty;
  return totalScore;
}

/**
 * Indexed Binary Max-Heap Priority Queue.
 *
 * Algorithmic Bounds:
 * - Enqueue: O(log N)
 * - ExtractMax: O(log N)
 * - PeekMax: O(1)
 * - RemoveById: O(log N) (via internal index map)
 * - Size, Severity Breakdown, Dwell-Sum: O(1)
 */
export class IndexedPriorityQueue {
  private heap: Ticket[] = [];
  private idToIndex: Map<string, number> = new Map();
  private severityCounts: Record<Severity, number> = { 1: 0, 2: 0, 3: 0, 4: 0 };
  private options: PriorityCalculatorOptions;

  constructor(options: PriorityCalculatorOptions = {}) {
    this.options = options;
  }

  public size(): number {
    return this.heap.length;
  }

  public isEmpty(): boolean {
    return this.heap.length === 0;
  }

  public getSeverityCounts(): Record<Severity, number> {
    return { ...this.severityCounts };
  }

  public contains(id: string): boolean {
    return this.idToIndex.has(id);
  }

  /**
   * Enqueue a ticket into the binary heap: O(log N)
   */
  public enqueue(ticket: Ticket, now: number = Date.now()): void {
    if (this.idToIndex.has(ticket.id)) {
      this.update(ticket, now);
      return;
    }

    ticket.lastDynamicScore = calculateDynamicScore(ticket, now, this.options);
    const index = this.heap.length;
    this.heap.push(ticket);
    this.idToIndex.set(ticket.id, index);
    this.severityCounts[ticket.severity]++;

    this.siftUp(index, now);
  }

  /**
   * Peek at the highest priority ticket without removing: O(1)
   */
  public peek(now: number = Date.now()): Ticket | null {
    if (this.heap.length === 0) return null;
    this.recalculateTopIfNeeded(now);
    return this.heap[0] ?? null;
  }

  /**
   * Extracts and returns the highest priority ticket: O(log N)
   */
  public extractMax(now: number = Date.now()): Ticket | null {
    if (this.heap.length === 0) return null;

    // Refresh scores for top elements to respect aging dynamics
    this.recalculateTopIfNeeded(now);

    const maxTicket = this.heap[0];
    this.swap(0, this.heap.length - 1);
    this.heap.pop();
    this.idToIndex.delete(maxTicket.id);
    this.severityCounts[maxTicket.severity]--;

    if (this.heap.length > 0) {
      this.siftDown(0, now);
    }

    return maxTicket;
  }

  /**
   * Removes a specific ticket by ID: O(log N)
   */
  public remove(id: string, now: number = Date.now()): Ticket | null {
    const index = this.idToIndex.get(id);
    if (index === undefined) return null;

    const removedTicket = this.heap[index];
    const lastIndex = this.heap.length - 1;

    if (index === lastIndex) {
      this.heap.pop();
      this.idToIndex.delete(id);
      this.severityCounts[removedTicket.severity]--;
      return removedTicket;
    }

    this.swap(index, lastIndex);
    this.heap.pop();
    this.idToIndex.delete(id);
    this.severityCounts[removedTicket.severity]--;

    // Rebalance from the replaced position
    this.siftDown(index, now);
    this.siftUp(index, now);

    return removedTicket;
  }

  /**
   * Updates an existing ticket's priority score and rebalances: O(log N)
   */
  public update(ticket: Ticket, now: number = Date.now()): void {
    const index = this.idToIndex.get(ticket.id);
    if (index === undefined) {
      this.enqueue(ticket, now);
      return;
    }

    this.heap[index] = ticket;
    ticket.lastDynamicScore = calculateDynamicScore(ticket, now, this.options);
    this.siftDown(index, now);
    this.siftUp(index, now);
  }

  /**
   * Peek a window of tickets ordered by current dynamic priority without modifying the heap: O(K log K)
   */
  public peekRange(limit: number, offset: number = 0, now: number = Date.now()): Ticket[] {
    const scoredList = this.heap.map((t) => {
      const score = calculateDynamicScore(t, now, this.options);
      t.lastDynamicScore = score;
      return { ticket: t, score };
    });

    // Sort descending by dynamic score
    scoredList.sort((a, b) => b.score - a.score);

    return scoredList.slice(offset, offset + limit).map((item) => item.ticket);
  }

  /**
   * Rebalances and updates scores across the entire heap: O(N)
   */
  public refreshScores(now: number = Date.now()): void {
    for (let i = 0; i < this.heap.length; i++) {
      this.heap[i].lastDynamicScore = calculateDynamicScore(this.heap[i], now, this.options);
    }
    // Heapify bottom-up O(N)
    for (let i = Math.floor(this.heap.length / 2) - 1; i >= 0; i--) {
      this.siftDown(i, now);
    }
  }

  /**
   * Returns total cumulative dwell time of currently queued tickets
   */
  public getTotalCurrentDwellMs(now: number = Date.now()): { totalDwellMs: number; oldestDwellMs: number } {
    let totalDwellMs = 0;
    let oldestDwellMs = 0;
    for (const t of this.heap) {
      const dwell = Math.max(0, now - t.timestamps.enqueuedAt);
      totalDwellMs += dwell;
      if (dwell > oldestDwellMs) {
        oldestDwellMs = dwell;
      }
    }
    return { totalDwellMs, oldestDwellMs };
  }

  public getAll(): Ticket[] {
    return [...this.heap];
  }

  // --- Internal Heap Mechanics ---

  private recalculateTopIfNeeded(now: number): void {
    if (this.heap.length <= 1) return;
    // Check if dynamic aging or backoff has altered relative order between root and children
    this.refreshScores(now);
  }

  private siftUp(index: number, now: number): void {
    let current = index;
    while (current > 0) {
      const parent = Math.floor((current - 1) / 2);
      const currentScore = this.getScore(this.heap[current], now);
      const parentScore = this.getScore(this.heap[parent], now);

      if (currentScore > parentScore) {
        this.swap(current, parent);
        current = parent;
      } else {
        break;
      }
    }
  }

  private siftDown(index: number, now: number): void {
    let current = index;
    const length = this.heap.length;

    while (true) {
      let largest = current;
      const left = 2 * current + 1;
      const right = 2 * current + 2;

      if (left < length && this.getScore(this.heap[left], now) > this.getScore(this.heap[largest], now)) {
        largest = left;
      }

      if (right < length && this.getScore(this.heap[right], now) > this.getScore(this.heap[largest], now)) {
        largest = right;
      }

      if (largest !== current) {
        this.swap(current, largest);
        current = largest;
      } else {
        break;
      }
    }
  }

  private getScore(ticket: Ticket, now: number): number {
    return calculateDynamicScore(ticket, now, this.options);
  }

  private swap(i: number, j: number): void {
    const temp = this.heap[i];
    this.heap[i] = this.heap[j];
    this.heap[j] = temp;
    this.idToIndex.set(this.heap[i].id, i);
    this.idToIndex.set(this.heap[j].id, j);
  }
}

/**
 * Asynchronous FIFO Mutex lock for atomic execution of critical sections.
 * Guarantees serial execution without race conditions across multiple
 * asynchronous workers competing for the queue.
 */
export class AsyncMutex {
  private queue: (() => void)[] = [];
  private locked = false;

  public async acquire(): Promise<() => void> {
    if (!this.locked) {
      this.locked = true;
      return () => this.release();
    }

    return new Promise<() => void>((resolve) => {
      this.queue.push(() => {
        resolve(() => this.release());
      });
    });
  }

  public release(): void {
    if (this.queue.length > 0) {
      const next = this.queue.shift();
      if (next) next();
    } else {
      this.locked = false;
    }
  }

  /**
   * Executes a callback within the protected atomic lock scope.
   */
  public async withLock<T>(fn: () => Promise<T> | T): Promise<T> {
    const unlock = await this.acquire();
    try {
      return await fn();
    } finally {
      unlock();
    }
  }

  public isLocked(): boolean {
    return this.locked;
  }

  public waitingCount(): number {
    return this.queue.length;
  }
}
